cmake_minimum_required(VERSION 3.20)
project(native_expert LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(pinocchio REQUIRED)
find_package(nlohmann_json REQUIRED)
find_library(BLAS_LIBRARY blas REQUIRED)
find_library(LAPACK_LIBRARY lapack REQUIRED)
include(native_bullet.cmake)
add_executable(expert src/main.cpp src/physics.cpp src/controller.cpp)
target_link_libraries(expert PRIVATE native_bullet pinocchio::pinocchio nlohmann_json::nlohmann_json ${BLAS_LIBRARY} ${LAPACK_LIBRARY})
target_compile_options(expert PRIVATE -O3 -ffp-contract=off)
add_executable(expert_compatible src/main.cpp src/physics.cpp src/controller.cpp)
target_include_directories(expert_compatible PRIVATE ${BULLET_SOURCE}/examples ${BULLET_SOURCE}/src)
list(GET CMAKE_PREFIX_PATH 0 ENV_PREFIX)
file(GLOB PYBULLET_BINARY "${ENV_PREFIX}/lib/python*/site-packages/pybullet.cpython-*.so")
if(APPLE)
  file(GLOB PYTHON_LIBRARY "${ENV_PREFIX}/lib/libpython3.*.dylib")
else()
  file(GLOB PYTHON_LIBRARY "${ENV_PREFIX}/lib/libpython3.*.so")
endif()
if(NOT PYBULLET_BINARY OR NOT PYTHON_LIBRARY)
  message(FATAL_ERROR "Pinned PyBullet and libpython were not found in ${ENV_PREFIX}")
endif()
list(GET PYBULLET_BINARY 0 PYBULLET_BINARY)
list(GET PYTHON_LIBRARY 0 PYTHON_LIBRARY)
target_compile_definitions(expert_compatible PRIVATE REFERENCE_ENGINE DEFAULT_LIBPYTHON="${PYTHON_LIBRARY}" DEFAULT_PYBULLET="${PYBULLET_BINARY}")
target_link_libraries(expert_compatible PRIVATE pinocchio::pinocchio nlohmann_json::nlohmann_json ${BLAS_LIBRARY} ${LAPACK_LIBRARY} ${CMAKE_DL_LIBS})
target_compile_options(expert_compatible PRIVATE -O3 -ffp-contract=off)
file(SHA256 "${PYBULLET_BINARY}" ENGINE_SHA)
set(SOURCE_DIGESTS "")
foreach(SOURCE src/main.cpp src/physics.cpp src/controller.cpp src/expert.hpp src/numerics.hpp src/bullet_api.hpp)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${SOURCE}" DIGEST)
  string(APPEND SOURCE_DIGESTS "${SOURCE}:${DIGEST};")
endforeach()
string(SHA256 COLLECTOR_SHA "${SOURCE_DIGESTS}")
set_source_files_properties(src/main.cpp PROPERTIES COMPILE_DEFINITIONS "COLLECTOR_SHA=\"${COLLECTOR_SHA}\";ENGINE_SHA=\"${ENGINE_SHA}\";COMPILER_VERSION=\"${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}\"")
